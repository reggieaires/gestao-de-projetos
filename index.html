const SPREADSHEET_ID = "1r0Igx2msmlV1x5gU2rb_mlOjo_5oRWnOrIPcUU348vY"; const ABA_PROJETOS = "projetos"; const ABA_CONFIG = "config"; function doGet(e) { const acao = e.parameter.acao || ""; let resultado; if (acao === "listar") resultado = listarProjetos(); else if (acao === "indicadores") resultado = getIndicadores(e); else resultado = { erro: "Ação não reconhecida." }; return buildResponse(resultado); } function doPost(e) { let payload; try { payload = JSON.parse(e.postData.contents); } catch(err) { return buildResponse({ erro: "JSON inválido." }); } const acao = payload.acao || ""; let resultado; if (acao === "cadastrar") resultado = cadastrarProjeto(payload); else if (acao === "editar") resultado = editarProjeto(payload); else if (acao === "excluir") resultado = excluirProjeto(payload); else if (acao === "status") resultado = atualizarStatus(payload); else resultado = { erro: "Ação não reconhecida." }; return buildResponse(resultado); } function buildResponse(data) { const output = ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); return output; } function getSheet(nome) { return SpreadsheetApp .openById(SPREADSHEET_ID) .getSheetByName(nome); } function gerarId() { const cfg = getSheet(ABA_CONFIG); const dados = cfg.getDataRange().getValues(); let linha = -1; let atual = 0; for (let i = 1; i < dados.length; i++) { if (dados[i][0] === "ultimo_id") { linha = i + 1; atual = parseInt(dados[i][1]) || 0; break; } } const novo = atual + 1; if (linha > 0) cfg.getRange(linha, 2).setValue(novo); else cfg.appendRow(["ultimo_id", novo]); return "PRJ-" + String(novo).padStart(3, "0"); } function hoje() { return Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd" ); } function projetosParaObjetos(valores) { const cabecalho = valores[0]; return valores.slice(1).map(row => { const obj = {}; cabecalho.forEach((col, i) => { obj[col] = row[i]; }); return obj; }); } function listarProjetos() { const sheet = getSheet(ABA_PROJETOS); const valores = sheet.getDataRange().getValues(); if (valores.length <= 1) return { projetos: [] }; const projetos = projetosParaObjetos(valores).reverse(); return { projetos }; } function getIndicadores(e) { const sheet = getSheet(ABA_PROJETOS); const valores = sheet.getDataRange().getValues(); if (valores.length <= 1) return indicadoresVazios(); let projetos = projetosParaObjetos(valores); const fStatus = e && e.parameter.status ? e.parameter.status : ""; const fPrioridade = e && e.parameter.prioridade ? e.parameter.prioridade : ""; const fResponsavel = e && e.parameter.responsavel ? e.parameter.responsavel : ""; if (fStatus) projetos = projetos.filter(p => p.status === fStatus); if (fPrioridade) projetos = projetos.filter(p => p.prioridade === fPrioridade); if (fResponsavel) projetos = projetos.filter(p => p.responsavel === fResponsavel); const dataHoje = hoje(); const total = projetos.length; const andamento = projetos.filter(p => p.status === "Em andamento").length; const concluido = projetos.filter(p => p.status === "Concluído").length; const revisao = projetos.filter(p => p.status === "Em revisão").length; const naoIniciado = projetos.filter(p => p.status === "Não iniciado").length; const atrasados = projetos.filter(p => p.status !== "Concluído" && p.data_entrega && String(p.data_entrega) < dataHoje ).length; const porPrioridade = { Alta: 0, Média: 0, Baixa: 0 }; projetos.forEach(p => { if (porPrioridade[p.prioridade] !== undefined) porPrioridade[p.prioridade]++; }); return { total, andamento, concluido, revisao, naoIniciado, atrasados, porPrioridade }; } function indicadoresVazios() { return { total: 0, andamento: 0, concluido: 0, revisao: 0, naoIniciado: 0, atrasados: 0, porPrioridade: { Alta: 0, Média: 0, Baixa: 0 } }; } function cadastrarProjeto(p) { const sheet = getSheet(ABA_PROJETOS); const id = gerarId(); const agora = hoje(); sheet.appendRow([ id, p.nome, p.responsavel, p.data_inicio, p.data_entrega, p.status, p.prioridade, p.descricao || "", agora, agora ]); return { sucesso: true, id }; } function editarProjeto(p) { const sheet = getSheet(ABA_PROJETOS); const valores = sheet.getDataRange().getValues(); for (let i = 1; i < valores.length; i++) { if (valores[i][0] === p.id) { const l = i + 1; sheet.getRange(l, 2).setValue(p.nome || valores[i][1]); sheet.getRange(l, 3).setValue(p.responsavel || valores[i][2]); sheet.getRange(l, 4).setValue(p.data_inicio || valores[i][3]); sheet.getRange(l, 5).setValue(p.data_entrega || valores[i][4]); sheet.getRange(l, 6).setValue(p.status || valores[i][5]); sheet.getRange(l, 7).setValue(p.prioridade || valores[i][6]); sheet.getRange(l, 8).setValue(p.descricao !== undefined ? p.descricao : valores[i][7]); sheet.getRange(l, 10).setValue(hoje()); return { sucesso: true }; } } return { erro: "Projeto não encontrado." }; } function excluirProjeto(p) { const sheet = getSheet(ABA_PROJETOS); const valores = sheet.getDataRange().getValues(); for (let i = 1; i < valores.length; i++) { if (valores[i][0] === p.id) { sheet.deleteRow(i + 1); return { sucesso: true }; } } return { erro: "Projeto não encontrado." }; } function atualizarStatus(p) { const sheet = getSheet(ABA_PROJETOS); const valores = sheet.getDataRange().getValues(); for (let i = 1; i < valores.length; i++) { if (valores[i][0] === p.id) { sheet.getRange(i + 1, 6).setValue(p.status); sheet.getRange(i + 1, 10).setValue(hoje()); return { sucesso: true }; } } return { erro: "Projeto não encontrado." }; }
